* Introduction

This is a keymap for the Corne (CRKBD) keyboard using the Dvorak
layout.  It is designed to minimize finger movement and maximize
comfort during extended typing sessions.

The keymap is inspired by the Miryoku keymap, but it utilizes all keys
on the Corne 46-key layout.  Miryoku focuses on a 34-key philosophy
with home-row modifiers and layers; this keymap extends that approach
to take advantage of the additional keys available on the Corne
keyboard.

Features included in this keymap:

- Home-row modifiers on the base layer
- Multiple layers for different functions (navigation, numbers,
  symbols, functions, media, mouse)
- Tap dance for quick access to boot, reset, lock, and caps functions
  (requiring double-tap to prevent accidental triggers)
- Caps word for typing SCREAMING_SNAKE_CASE without holding shift
- Auto shift for automatic shifting of keys when held longer
- Flow tap for improved tap-hold disambiguation on rolling keys
- Speculative hold for immediate modifier activation on home-row keys
- Secure lock to prevent unexpected keyboard usage
- RGB matrix with typing heatmap visualization

* Keymap layers

The keymap consists of several layers, each designed for a specific
purpose.  The base layer uses Dvorak layout with home-row modifiers,
while additional layers provide access to numbers, symbols,
navigation keys, function keys, media controls, and mouse buttons.

All layers except the base and lock layers are accessed through
layer-tap keys on the base layer or through momentary layer switches.
The lock layer can only be activated by the secure lock feature.

** The base layer

modifier keys are placed on both halves: the left hand uses left-hand
modifiers (LGUI, LALT, LCTL, LSFT) while the right hand uses
right-hand modifiers (RSFT, RCTL, LALT, RGUI), allowing access to
Control, Alt, Shift, and GUI without leaving the home row.

The thumb keys provide access to Space, Enter, Backspace, Delete,
Tab, and layer switches.

** Handle shifted keys

The RGB matrix keys on the media layer have alternate actions when
combined with Shift.  This allows increasing and decreasing values
without requiring a separate layer.

For example, holding Shift and pressing RM_HUEU will decrease the hue
instead of increasing it.

** Caps word

Caps word is a feature that enables typing in ALL_CAPS without holding
down the Shift key.  When activated, alphabetic characters are
automatically shifted.  The feature automatically deactivates when a
non-continuation key is pressed.

#+begin_src c :tangle yes :noweb yes
  bool caps_word_press_user(uint16_t keycode) {
    switch (keycode) {
  #define X(x) case x:
      <<table-to-xlist(table=dvorak-alphabets)>>
  #undef X
    case KC_MINS:
      add_weak_mods(MOD_BIT(KC_LSFT));
      return true;

    case KC_1 ... KC_9:
    case KC_UNDS:
    case KC_BSPC:
    case KC_DEL:
      return true;

    default:
      return false;
    }
  }
#+end_src

** Tap dance

Tap dance allows a single key to have different actions based on how
many times it is tapped.  The following tap dances are defined:

- U_BOOT :: Double-tap to jump to bootloader
- U_RESET :: Double-tap to reset the keyboard
- U_LKSCRN :: Double-tap to lock the screen
- U_LOCKKB :: Double-tap to lock the keyboard (secure lock)
- U_CAPS :: Single-tap toggles caps word, double-tap toggles Caps Lock

#+name: tap-dances
| Identifier | Action                                    |
|------------+-------------------------------------------|
| U_BOOT     | ACTION_TAP_DANCE_FN(dance_boot)           |
| U_RESET    | ACTION_TAP_DANCE_FN(dance_reset)          |
| U_LKSCRN   | ACTION_TAP_DANCE_DOUBLE(KC_NO, LAG(KC_L)) |
| U_LOCKKB   | ACTION_TAP_DANCE_FN(dance_lock)           |
| U_CAPS     | ACTION_TAP_DANCE_FN(dance_caps)           |


#+begin_src c :tangle yes
  static void dance_boot(tap_dance_state_t *state, void *user_data) {
    if (state->count >= 2) {
      bootloader_jump();
    }
  }

  static void dance_reset(tap_dance_state_t *state, void *user_data) {
    if (state->count >= 2) {
      mcu_reset();
    }
  }

  static void dance_lock(tap_dance_state_t *state, void *user_data) {
    if (state->count == 2) {
      layer_on(L_LOCK);
      rgb_matrix_disable_noeeprom();
    } if (state->count == 3) {
      rgb_matrix_enable_noeeprom();
      layer_off(L_LOCK);
    }
  }

  static void dance_caps(tap_dance_state_t *state, void *user_data) {
    if (state->count == 1) {
      caps_word_toggle();
    } else {
      tap_code(KC_CAPS);
    }
  }
#+end_src

** Auto shift

Auto shift automatically sends the shifted version of a key when it
is held longer than ~AUTO_SHIFT_TIMEOUT~.  The shifted versions of
Dvorak symbol keys, digits, Tab, and the semicolon layer-tap key
are enabled.

RETRO_SHIFT is also enabled.  For tap-hold keys that are in the
auto-shift list (such as ~LT(L_BUTTON, KC_SCLN)~), holding and
releasing the key without pressing another key will produce the
shifted version of the tap keycode, as long as the hold time is
within the ~RETRO_SHIFT~ window.  Holds longer than that trigger
the normal hold action instead.

#+begin_src c :tangle yes :noweb yes
  bool get_auto_shifted_key(uint16_t keycode, keyrecord_t *record) {
    switch (keycode) {
    case KC_1 ... KC_0:
    case KC_TAB:
  #define X(x)  case x:
      <<table-to-xlist(table=dvorak-symbols)>>
  #undef X
    case LT(L_BUTTON, KC_SCLN):
      return true;
    default:
      return false;
    }
  }
#+end_src

** Flow tap

Flow tap allows Space and Dvorak symbol keys to be used for tap-based
hold/tap disambiguation.  It is disabled when modifiers or Alt are
held to avoid interfering with hotkeys.

#+begin_src c :tangle yes :noweb yes
  bool is_flow_tap_key(uint16_t keycode) {
    if ((get_mods() & (MOD_MASK_CG | MOD_BIT_LALT)) != 0) {
      return false;
    }
    switch (get_tap_keycode(keycode)) {
    case KC_SPC:
  #define X(x)  case x:
      <<table-to-xlist(table=dvorak-alphabets)>>
      <<table-to-xlist(table=dvorak-symbols)>>
  #undef X
      return true;
    }
    return false;
  }
#+end_src

** Chordal hold (Disabled)

Chordal hold is disabled by default because it prevents single-handed
shortcuts.  When enabled, if a mod-tap key is pressed and then another
key on the same hand is pressed within the tapping term, the mod-tap
key is settled as tapped instead of held, which breaks shortcuts that
use modifiers and keys on the same hand.

If your typing style does not rely on single-handed shortcuts, you can
re-enable it by adding the following to your =config.h=:

#+begin_src c
  #define CHORDAL_HOLD
#+end_src

** Speculative hold (Disabled)

Speculative hold is disabled because it causes issues with Chinese
IMEs.  When a home-row modifier key is pressed, speculative hold
immediately sends the modifier (e.g., Shift), which Chinese IMEs
interpret as an English/Chinese mode switch.  This results in
unintended language switching while typing.

If you do not use Chinese IMEs or other input methods that rely on
modifier keys for mode switching, you can re-enable it by adding the
following to your =config.h=:

#+begin_src c
  #define SPECULATIVE_HOLD
#+end_src

* Scripts

This section contains helper scripts used by org-babel to convert
tables into C code and Makefile variables.  These are internal
utilities and are not part of the firmware itself.

#+name: table-example
| Identifier | Value |
|------------+-------|
| FOO        |     1 |
| BAR        |       |
| BAZ        |     2 |

#+name: table-to-define
#+begin_src bash :var table=table-example :var prefix="" :results output :wrap src c
  if [[ "${table@a}" = *"A"* ]]; then
      for id in "${!table[@]}"; do
  	echo "#define" "$prefix$id" "${table[$id]}"
      done
  else
      for id in "${table[@]}"; do
  	echo "#define" "$prefix$id"
      done
  fi
#+end_src

#+results: table-to-define
#+begin_src c
#define FOO 1
#define BAR
#define BAZ 2
#+end_src

#+name: table-to-var
#+begin_src bash :var table=table-example :var prefix="" :results output :wrap src makefile
  for id in "${!table[@]}"; do
      echo "$prefix$id" = "${table[$id]}"
  done
#+end_src

#+results: table-to-var
#+begin_src makefile
FOO = 1
BAR =
BAZ = 2
#+end_src

#+name: table-to-enum
#+begin_src bash :var table=table-example :var name="" :var prefix="" :results output :wrap src c
  echo enum $name '{'
  if [[ "${table@a}" = *"A"* ]]; then
      for id in "${!table[@]}"; do
    	echo "  $prefix$id${table[$id]:+ = ${table[$id]}},"
      done
  else
      for id in "${table[@]}"; do
  	echo " $prefix$id,"
      done
  fi
  echo '};'
#+end_src

#+results: table-to-enum
#+begin_src c
enum {
  FOO = 1,
  BAR,
  BAZ = 2,
};
#+end_src

#+name: table-to-xlist
#+begin_src elisp :var table=table-example :wrap src c
  (cl-loop for row in table
  	 concat (format "X(%s)\n"
  			(string-join (mapcar (lambda (x) (format "%s" x))
  					     row)
  			     ", ")))
#+end_src

#+results: table-to-xlist
#+begin_src c
X(FOO, 1)
X(BAR, )
X(BAZ, 2)
#+end_src
